## 答案：

import turtle as t
import math
def yuanhu(r, n):
    # 用 循环绘制
    # for i in range(0,n):
    #     t.fd(0.0174*r)
    #     t.lt(1)
    # 用 circle 方法绘制
    t.circle(r, n)

r=40
n=6
m=100

for i in range(0, n):
    yuanhu(r, m)
    t.rt(m)
    t.rt(360/n)
