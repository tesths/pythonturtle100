## 解析：

绘制由叶子组成的图案。

## 答案：

```python
import turtle as t
def yuanhu(r, n):
    # 用 循环绘制
    # for i in range(0,n):
    #     t.fd(0.0174*r)
    #     t.rt(1)
    # 用 circle 方法绘制
    t.circle(-r, n)
 
def yezi(r, n):
    for i in range(0,2):
        yuanhu(r,n)
        t.rt(180-n)
# 3 对树叶
n=3
# 4 对树叶
# n=4
t.lt(90)
for i in range(0, n):
    t.fd(20)
    t.lt(90)
    yezi(20, 90)
    t.rt(90)
    yezi(20, 90)

t.fd(20)
t.lt(45)
yezi(20, 90)
```
