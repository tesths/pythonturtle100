## 解析：

绘制叶子的组合。

## 答案：

```python
import turtle as t
import math
def yuanhu(r, n):
    # 用 循环绘制
    # for i in range(0,n):
    #     t.fd(0.0174*r)
    #     t.rt(1)
    # 用 circle 方法绘制
    t.circle(-r, n)

def yezi(r, n):
    for i in range(0,2):
        yuanhu(r,n)
        t.rt(180-n)
    return
 
r=60
m=90
for i in range(0, 4):
    yezi(r,m)
    t.rt(90)
t.rt(45)
for i in range(0, 4):
    # 计算使用
    # print(r*math.sin(math.radians(m/2))*2)
    t.fd(85)
    t.bk(85)
    t.rt(90)
```
